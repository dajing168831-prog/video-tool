"""H3 创作模式「拆槽层」的纯逻辑自检（**不发网络请求、不花钱**）。

被测对象：`app/services/h3_base_slots.py` —— 把顾客的话变成编译器认的 spec。
这里全部用**桩模型**（返回写死的 JSON），所以只验证「拆槽层自己的规矩」：

  · 模型给了越界值 → 收口（丢掉/夹紧）并如实报出来，不猜、不静默
  · 顾客的原话 → 数字与引号里的招牌文字一个不许丢
  · 切点 → 缺的均分、乱的调正、超时长的报出来（官方 `base-en:86`）
  · 最终产出 → 过 `ruler_h3_base` 的 22 条判据，红 0

🩸 为什么必须跑尺子：拆槽层绿了不代表提示词能用。只测"函数返回了 dict"
等于自己给自己发合格证 —— 终点是**尺子不红**。

运行：python scripts/test_h3_base_slots.py --model h3
"""
from __future__ import annotations

import sys
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parents[1]))
sys.path.insert(0, str(Path(__file__).resolve().parent))
sys.stdout.reconfigure(encoding="utf-8", errors="replace")

import _model_lock  # noqa: E402

RESULT: list[tuple[bool, str, str]] = []


def ok(name: str, detail: str = "") -> None:
    RESULT.append((True, name, detail))


def bad(name: str, detail: str) -> None:
    RESULT.append((False, name, detail))


def expect(cond: bool, name: str, detail: str = "") -> bool:
    (ok if cond else bad)(name, "" if cond else detail)
    return cond


def stub(payload):
    """桩模型：不管问什么都返回同一份 JSON（零花费）。"""
    def _chat(sys_msg: str, user_msg: str) -> dict:
        _chat.seen.append((sys_msg, user_msg))
        return payload
    _chat.seen = []
    return _chat


def run(name: str, fn) -> None:
    try:
        fn()
    except Exception as e:                                   # noqa: BLE001
        bad(name, f"抛异常：{type(e).__name__}: {e}")

def _rules_red(prompt: str, ctx: dict) -> list:
    import ruler_h3_base as R
    return [r for r in R.check(prompt, ctx) if r[1] == "红"]

    from app.services import h3_base_slots as S
    for kw, name in (({"text": "  "}, "20a 空输入"),
                     ({"text": "x", "kind": "weird"}, "20b 未知 kind")):

        text = kw.pop("text")

            S.extract(text, chat=stub({}), **kw)

        else:
            bad(name + " → 报错", "没报错")


def t_no_key_no_network() -> None:
    """没配 key 时必须**当场报错**，不许静默走网络（桩之外唯一的联网路径）。"""
    from app.services import h3_base_slots as S
    from app.core.config import settings

    if not (settings.ark_api_key or "").strip():
        ok("21a 未配 key → 报错（本机本来就没配，直接验）")
        return
    old = settings.ark_api_key
    try:
        object.__setattr__(settings, "ark_api_key", "")   # settings 是 frozen 的
        try:
            S._default_chat("s", "u")                     # 必须在发请求之前就抛
        except S.SlotError:
            ok("21a 未配 key → 报错（没发请求）")
        else:
            bad("21a 未配 key → 报错（没发请求）", "居然没报错")
    finally:
        object.__setattr__(settings, "ark_api_key", old)


def t_ruler_selfprove() -> None:
    """🩸 尺子自证：上面那几条「过尺子红 0」只有在尺子**会响**时才有意义。

    拿同一份合法产出，故意改坏三处，尺子必须各报一次红；坏样本不响 = 判据是空的。
    """
    from app.services.models import h3_base_compile as H

    good = H.compile({"mode": "t2va", "duration": 5.0, "ratio": "16:9",
                      "style": "Live-action, natural light",
                      "shots": [{"body": "A woman walks down the street.",
                                 "camera": {"type": "Push In",
                                            "amplitude": "small",
                                            "speed": "slow"}}],
                      "soundscape": "Wind moves through the trees.",
                      "music": ""})["text"]
    expect(not _rules_red(good, {"mode": "t2va"}),
           "22a 好样本不响", str(_rules_red(good, {"mode": "t2va"}))[:300])

    probes = {
        "22b 正文中文化 → B3": good.replace("A woman walks down the street.",
                                            "一个女人走过街道。"),
        "22c 自造运镜 → B12": good.replace("pushes in", "dollies in"),
        "22d 删掉 music 字段 → B1": good.split("non_diegetic_music:")[0].rstrip(),
    }
    for name, p in probes.items():
        expect(bool(_rules_red(p, {"mode": "t2va"})), name, "尺子没响（假绿）")


def main() -> int:
    if "--model" not in sys.argv:
        print("⛔ 必须点名型号：--model h3")
        return 2
    i = sys.argv.index("--model")
    sel = sys.argv[i + 1] if i + 1 < len(sys.argv) else ""
    if sel != "h3":
        print("⛔ 本文件只装 H3 的断言（一轮一家模型）：--model h3")
        return 2
    _model_lock.guard("h3")

    cases = [
        ("1 运镜越界", t_camera_illegal),
        ("2 幅度速度越界", t_amp_speed_illegal),
        ("3 合法运镜", t_camera_legal),
        ("4 镜头数上限", t_too_many_shots),
        ("5 空输入形状", t_empty_shots),
        ("6 切点均分", t_cut_auto),
        ("7 切点递增", t_cut_order),
        ("8 切点超时长", t_cut_over),
        ("9 数字保真", t_keep_number),
        ("10 招牌保真", t_keep_sign),
        ("11 序数不误报", t_ordinal_noise),
        ("12 汉字告警", t_cjk_warning),
        ("13 语言识别", t_dialogue),
        ("14 画外音", t_voiceover),
        ("15 T2VA 端到端", t_e2e_t2va),
        ("16 I2VA 端到端", t_e2e_i2va),
        ("17 多镜端到端", t_e2e_multi),
        ("18 护栏在场", t_idea_guard),
        ("19 user 消息", t_user_message),
        ("20 入参校验", t_bad_input),
        ("21 未配 key", t_no_key_no_network),
        ("22 尺子自证", t_ruler_selfprove),
    ]
    for name, fn in cases:
        run(name, fn)

    passed = sum(1 for r in RESULT if r[0])
    for good, name, detail in RESULT:
        if good:
            print(f"  ✅ {name}")
        else:
            print(f"  ❌ {name}")
            if detail:
                print(f"       {detail}")
    print()
    print(f"拆槽层自检：通过 {passed} / {len(RESULT)}，红 {len(RESULT) - passed}")
    return 0 if passed == len(RESULT) else 1


if __name__ == "__main__":
    sys.exit(main())
